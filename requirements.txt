# Dependências de execução do ScreenShare 2.0

# fixar versões exatas quebraria a instalação nas versões mais novas do

# Captura de tela, conversão de imagem e interface de vídeo.
mss>=9.0.1

# pelo servidor de sinalização incluído no repositório.
aiortc>=1.15.0
av>=17.1.0
aiohttp>=3.14.3

# Áudio: opcional, porém recomendado.
# O sounddevice é o motor preferencial por ter rodas compatíveis com o

#   Linux (Debian/Ubuntu):  sudo apt install libportaudio2
# Se o sounddevice não estiver presente, o projeto tenta o PyAudio
# automaticamente; sem nenhum dos dois, roda apenas com vídeo e chat.
sounddevice>=0.4.6
